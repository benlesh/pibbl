import { createRequire } from 'node:module';

const require = createRequire(import.meta.url);
const { chromium } = require('/Users/benlesh/GitHub/crc/node_modules/playwright');

const browser = await chromium.launch({ headless: true });
const page = await browser.newPage();
try {
  await page.goto('http://127.0.0.1:5199/', { waitUntil: 'networkidle' });
  const result = await page.evaluate(async () => {
    const root = '/@fs/Users/benlesh/GitHub/crc';
    const core = await import(`${root}/packages/core/src/index.ts`);
    const physics = await import(`${root}/packages/core/src/features/physics/2d.ts`);
    const controllerModule = await import(
      `${root}/packages/core/src/features/physics/lib/2d/runtime/controller.ts`
    );
    const scheduler = await import(
      `${root}/packages/core/src/lib/scheduler/testing.ts`
    );
    const driver = new scheduler.ManualSchedulerDriver();
    const restore = scheduler.installSchedulerDriverForTest(driver);
    const canvas = document.createElement('canvas');
    canvas.width = 80;
    canvas.height = 80;
    document.body.append(canvas);
    const present = core.signal(false);
    let world;
    function Capture() {
      world = physics.crcPhysicsWorld2D();
      return undefined;
    }
    function Scene() {
      return present.get()
        ? core.createElement(core.Circle, {
            key: `body-${Date.now()}`,
            physics: physics.staticBody2D(),
            style: { cx: 20, cy: 20, radius: 5 },
          })
        : undefined;
    }
    const mount = core.crc(
      canvas,
      core.createElement(physics.PhysicsWorld2D, {
        children: [core.createElement(Capture, {}), core.createElement(Scene, {})],
      }),
    );
    let time = 0;
    const flush = () => {
      for (let guard = 0; guard < 8 && driver.pendingHostFrames > 0; guard += 1) {
        driver.flush((time += 1));
      }
    };
    flush();
    const controller = controllerModule.requirePhysicsWorld2DController(world);
    const before = {
      bodies: controller.materializedBodyHandles.size,
      colliders: controller.materializedColliderHandles.size,
    };
    for (let cycle = 0; cycle < 100; cycle += 1) {
      present.set(true);
      flush();
      const hit = physics.queryPoint2D(world, [20, 20]);
      if (!hit) throw new Error(`missing query hit at cycle ${cycle}`);
      present.set(false);
      flush();
    }
    const afterCycles = {
      bodies: controller.materializedBodyHandles.size,
      colliders: controller.materializedColliderHandles.size,
    };
    mount.dispose();
    const afterDispose = {
      bodies: controller.materializedBodyHandles.size,
      colliders: controller.materializedColliderHandles.size,
    };
    restore();
    canvas.remove();
    return { before, afterCycles, afterDispose, pendingFrames: driver.pendingHostFrames };
  });
  console.log(JSON.stringify(result));
} finally {
  await browser.close();
}
