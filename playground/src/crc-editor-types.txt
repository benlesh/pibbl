  export type Canvas = HTMLCanvasElement | OffscreenCanvas;
  export type RenderingContext2D = CanvasRenderingContext2D | OffscreenCanvasRenderingContext2D;
  export type FalsyNodes = undefined | false | null | '' | 0;
  export type CRCNode = CompEl<any, any> | void | FalsyNodes;

  export type CRCEventPhase = 1 | 2 | 3;
  export interface CRCEventTarget {
    setPointerCapture(pointerId: number): void;
    hasPointerCapture(pointerId: number): boolean;
    releasePointerCapture(pointerId: number): void;
  }
  export interface CRCEvent<TOriginal extends Event = Event> {
    readonly type: string;
    readonly target: CRCEventTarget;
    readonly currentTarget: CRCEventTarget;
    readonly relatedTarget: CRCEventTarget | null;
    readonly eventPhase: CRCEventPhase;
    readonly bubbles: boolean;
    readonly cancelable: boolean;
    readonly defaultPrevented: boolean;
    readonly timeStamp: number;
    readonly originalEvent: TOriginal;
    readonly sourceEvent?: Event;
    composedPath(): readonly CRCEventTarget[];
    preventDefault(): void;
    stopPropagation(): void;
    stopImmediatePropagation(): void;
  }
  export interface CRCEventTargetSnapshot { readonly isFocused: boolean; readonly isFocusVisible: boolean }
  export interface CRCEventTargetOptions {
    readonly path: Path2D;
    readonly fill?: boolean;
    readonly lineInteractionWidth?: number;
    readonly cursor?: string;
    readonly keyboardFocusable?: boolean;
    readonly keyboardNavigationBounds?: Readonly<LayoutBox>;
  }
  interface CRCModifierEvent<TOriginal extends Event> extends CRCEvent<TOriginal> {
    readonly altKey: boolean;
    readonly ctrlKey: boolean;
    readonly metaKey: boolean;
    readonly shiftKey: boolean;
  }
  export interface CRCMouseEvent<TOriginal extends MouseEvent = MouseEvent> extends CRCModifierEvent<TOriginal> {
    readonly x: number; readonly y: number;
    readonly clientX: number; readonly clientY: number;
    readonly screenX: number; readonly screenY: number;
    readonly movementX: number; readonly movementY: number;
    readonly button: number; readonly buttons: number; readonly detail: number;
  }
  export interface CRCPointerEvent extends CRCMouseEvent<PointerEvent> {
    readonly pointerId: number; readonly width: number; readonly height: number;
    readonly pressure: number; readonly tangentialPressure: number;
    readonly tiltX: number; readonly tiltY: number; readonly twist: number;
    readonly pointerType: string; readonly isPrimary: boolean;
  }
  export interface CRCTouchEvent extends CRCModifierEvent<TouchEvent> {
    readonly x: number; readonly y: number;
    readonly touches: TouchList; readonly targetTouches: TouchList; readonly changedTouches: TouchList;
  }
  export interface CRCWheelEvent extends CRCMouseEvent<WheelEvent> {
    readonly deltaX: number; readonly deltaY: number; readonly deltaZ: number; readonly deltaMode: number;
  }
  export interface CRCKeyboardEvent extends CRCModifierEvent<KeyboardEvent> {
    readonly key: string; readonly code: string; readonly location: number;
    readonly repeat: boolean; readonly isComposing: boolean;
  }
  export interface CRCFocusEvent extends CRCEvent<FocusEvent> {}
  export interface CRCInputEvent extends CRCEvent<InputEvent> {
    readonly data: string | null; readonly dataTransfer: DataTransfer | null;
    readonly inputType: string; readonly isComposing: boolean;
  }
  export interface CRCCompositionEvent extends CRCEvent<CompositionEvent> { readonly data: string }
  export interface CRCClipboardEvent extends CRCEvent<ClipboardEvent> { readonly clipboardData: DataTransfer | null }
  export interface CRCDragEvent extends CRCMouseEvent<DragEvent> { readonly dataTransfer: DataTransfer | null }
  export interface CRCEventMap {
    pointerover: CRCPointerEvent; pointerenter: CRCPointerEvent; pointerdown: CRCPointerEvent;
    pointermove: CRCPointerEvent; pointerrawupdate: CRCPointerEvent; pointerup: CRCPointerEvent;
    pointercancel: CRCPointerEvent; pointerout: CRCPointerEvent; pointerleave: CRCPointerEvent;
    gotpointercapture: CRCPointerEvent; lostpointercapture: CRCPointerEvent;
    click: CRCMouseEvent; dblclick: CRCMouseEvent; auxclick: CRCMouseEvent; contextmenu: CRCMouseEvent;
    mouseover: CRCMouseEvent; mouseenter: CRCMouseEvent; mousedown: CRCMouseEvent;
    mousemove: CRCMouseEvent; mouseup: CRCMouseEvent; mouseout: CRCMouseEvent; mouseleave: CRCMouseEvent;
    touchstart: CRCTouchEvent; touchmove: CRCTouchEvent; touchend: CRCTouchEvent; touchcancel: CRCTouchEvent;
    wheel: CRCWheelEvent; keydown: CRCKeyboardEvent; keyup: CRCKeyboardEvent;
    focus: CRCFocusEvent; blur: CRCFocusEvent; focusin: CRCFocusEvent; focusout: CRCFocusEvent;
    beforeinput: CRCInputEvent; input: CRCInputEvent; change: CRCEvent;
    compositionstart: CRCCompositionEvent; compositionupdate: CRCCompositionEvent; compositionend: CRCCompositionEvent;
    select: CRCEvent; selectionchange: CRCEvent;
    copy: CRCClipboardEvent; cut: CRCClipboardEvent; paste: CRCClipboardEvent;
    dragstart: CRCDragEvent; drag: CRCDragEvent; dragend: CRCDragEvent;
    dragenter: CRCDragEvent; dragleave: CRCDragEvent; dragover: CRCDragEvent; drop: CRCDragEvent;
  }
  export type CRCEventType = keyof CRCEventMap;
  interface CRCEventPropMap {
    pointerover: 'onPointerOver'; pointerenter: 'onPointerEnter'; pointerdown: 'onPointerDown';
    pointermove: 'onPointerMove'; pointerrawupdate: 'onPointerRawUpdate'; pointerup: 'onPointerUp';
    pointercancel: 'onPointerCancel'; pointerout: 'onPointerOut'; pointerleave: 'onPointerLeave';
    gotpointercapture: 'onGotPointerCapture'; lostpointercapture: 'onLostPointerCapture';
    click: 'onClick'; dblclick: 'onDblClick'; auxclick: 'onAuxClick'; contextmenu: 'onContextMenu';
    mouseover: 'onMouseOver'; mouseenter: 'onMouseEnter'; mousedown: 'onMouseDown';

  }
  export type CRCEventPropName = CRCEventPropMap[CRCEventType];
  export type CRCEventCapturePropName = `${CRCEventPropName}Capture`;
  export type CRCEventHandler<K extends CRCEventType> = (event: CRCEventMap[K]) => void;
  export type CRCEventHandlers = Partial<{
    [K in CRCEventType as CRCEventPropMap[K]]: CRCEventHandler<K>;
  } & {
    [K in CRCEventType as `${CRCEventPropMap[K]}Capture`]: CRCEventHandler<K>;
  }>;

  export interface SystemStyle { custom?: unknown }
  type ReservedElementKey = 'key' | 'style';
  type UnionKeys<T> = T extends unknown ? keyof T : never;
  type ProgramProps<P> = 0 extends (1 & P) ? P : Extract<UnionKeys<P>, ReservedElementKey> extends never ? P : never;
  type RequiredKeys<T> = T extends unknown ? { [K in keyof T]-?: Pick<T, K> extends Required<Pick<T, K>> ? K : never }[keyof T] : never;
  type StyleField<S> = [RequiredKeys<S>] extends [never] ? { style?: S } : { style: S };
  type AllocationValue = `${number}%` | 'auto';
  type DeclaredStyleValue<S> = S extends unknown ? S[Exclude<keyof S, 'custom'>] : never;
  type RequiresResolver<S> = [Extract<DeclaredStyleValue<S>, AllocationValue>] extends [never] ? false : true;
  export type SpecifiedStyle<S extends SystemStyle> = Readonly<S>;
  export type NormalizedStyle<S extends SystemStyle> = Readonly<S>;
  export type ResolvedStyle<S extends SystemStyle> = Readonly<S>;
  export interface CompEl<P = any, S extends SystemStyle = SystemStyle, N extends SystemStyle = S, R extends SystemStyle = N> {
    type: CompFn<P, S, N, R>;
    props: ProgramProps<P>;
    style: SpecifiedStyle<S>;
    key?: string;
  }

  export type ComponentInput<P, S extends SystemStyle = SystemStyle> = ProgramProps<P> & { key?: string } & StyleField<S>;
  export interface ComponentCapabilities<S extends SystemStyle, N extends SystemStyle = S, R extends SystemStyle = N, P = unknown> {
    normalizeStyle?: (style: SpecifiedStyle<S>) => NormalizedStyle<N>;
    resolveStyle?: (style: NormalizedStyle<N>, context: Readonly<StyleResolutionContext>) => ResolvedStyle<R>;
    measure?: (input: MeasureInput<ProgramProps<P>, N>) => MeasureResult;
  }
  type CapabilityArguments<P, S extends SystemStyle, N extends SystemStyle, R extends SystemStyle> = RequiresResolver<R> extends true ? [capabilities: never] : RequiresResolver<S> extends true ? [capabilities: Readonly<ComponentCapabilities<S, N, R, P>> & { resolveStyle: NonNullable<ComponentCapabilities<S, N, R, P>['resolveStyle']> }] : [capabilities?: Readonly<ComponentCapabilities<S, N, R, P>>];
  export type CompFn<P, S extends SystemStyle = SystemStyle, N extends SystemStyle = S, R extends SystemStyle = N> = ((props: ProgramProps<P>, style: ResolvedStyle<R>, context: RenderingContext2D) => CRCNode[] | CRCNode | void) & Readonly<ComponentCapabilities<S, N, R, P>>;
  export type PixelGridAlignment = 'none' | 'round' | 'ceil' | 'floor';
  export type FillStyle = CanvasRenderingContext2D['fillStyle'];
  export type StrokeStyle = CanvasRenderingContext2D['strokeStyle'];

  export type Percentage = `${number}%`;
  export type Length = number | Percentage | 'auto';
  export type EdgeValues = number | Readonly<{ top?: number; right?: number; bottom?: number; left?: number }>;
  export interface Size { width: number; height: number }
  export interface LayoutBox extends Size { x: number; y: number }
  export interface Edges { top: number; right: number; bottom: number; left: number }
  export interface Constraints { minWidth: number; maxWidth: number; minHeight: number; maxHeight: number }
  export interface StyleResolutionContext { allocation: Readonly<LayoutBox>; percentageBasis: Readonly<LayoutBox>; constraints: Readonly<Constraints>; component: string; algorithm: 'render-dispatch' }
  export interface BoxStyle extends SystemStyle {
    width?: Length; height?: Length;
    minWidth?: Exclude<Length, 'auto'>; maxWidth?: Exclude<Length, 'auto'>;
    minHeight?: Exclude<Length, 'auto'>; maxHeight?: Exclude<Length, 'auto'>;
    padding?: EdgeValues; overflow?: 'visible' | 'clip';
  }
  export interface ResolvedBoxStyle { borderBox: LayoutBox; contentBox: LayoutBox }
  export interface LayoutItemStyle {
    left?: Exclude<Length, 'auto'>; top?: Exclude<Length, 'auto'>;
    alignSelf?: 'auto' | 'start' | 'end' | 'center' | 'stretch' | 'flex-start' | 'flex-end';
    justifySelf?: 'auto' | 'start' | 'end' | 'center' | 'stretch';
    flexBasis?: Length; flexGrow?: number; flexShrink?: number;
    gridColumnStart?: number; gridColumnSpan?: number; gridRowStart?: number; gridRowSpan?: number;
  }
  export type FlexItemStyle = Pick<LayoutItemStyle, 'flexBasis' | 'flexGrow' | 'flexShrink'> & { alignSelf?: 'auto' | 'stretch' | 'flex-start' | 'flex-end' | 'center' };
  export interface GridItemStyle extends SystemStyle {
    gridColumnStart?: number; gridColumnSpan?: number; gridRowStart?: number; gridRowSpan?: number;
    alignSelf?: 'auto' | 'stretch' | 'start' | 'center' | 'end';
    justifySelf?: 'auto' | 'stretch' | 'start' | 'center' | 'end';
  }
  export interface PaintStyle extends SystemStyle { fill?: FillStyle; stroke?: StrokeStyle; strokeWidth?: number; opacity?: number }
  export interface CursorStyle extends SystemStyle { cursor?: string }
  export interface PathStyle extends PaintStyle, CursorStyle, BoxStyle, LayoutItemStyle { d: Path2D; lineInteractionWidth?: number }
  export interface RectStyle extends PaintStyle, CursorStyle, BoxStyle, LayoutItemStyle { x?: Exclude<Length, 'auto'>; y?: Exclude<Length, 'auto'>; width: Length; height: Length; alignToPixelGrid?: PixelGridAlignment; lineInteractionWidth?: number }
  export interface SvgPathStyle extends PaintStyle, CursorStyle, BoxStyle, LayoutItemStyle { d: string; lineInteractionWidth?: number }
  export interface LineStyle extends SystemStyle, CursorStyle, BoxStyle, LayoutItemStyle { coords: readonly (readonly [number, number])[]; stroke?: StrokeStyle; strokeWidth?: number; opacity?: number; lineInteractionWidth?: number }
  export interface HorizontalLineStyle extends SystemStyle, CursorStyle, BoxStyle, LayoutItemStyle { y: Exclude<Length, 'auto'>; x1?: Exclude<Length, 'auto'>; x2?: Exclude<Length, 'auto'>; stroke?: StrokeStyle; strokeWidth?: number; opacity?: number; alignToPixelGrid?: PixelGridAlignment; lineInteractionWidth?: number }
  export interface VerticalLineStyle extends SystemStyle, CursorStyle, BoxStyle, LayoutItemStyle { x: Exclude<Length, 'auto'>; y1?: Exclude<Length, 'auto'>; y2?: Exclude<Length, 'auto'>; stroke?: StrokeStyle; strokeWidth?: number; opacity?: number; alignToPixelGrid?: PixelGridAlignment; lineInteractionWidth?: number }
  export type TextFit = 'visible' | 'squish' | 'ellipsis' | 'clip';
  export interface TextStyle extends PaintStyle, CursorStyle, BoxStyle, LayoutItemStyle { text: string; x?: Exclude<Length, 'auto'>; y?: Exclude<Length, 'auto'>; lineHeight?: number; wrap?: boolean; fit?: TextFit; font?: string; textBaseline?: CanvasTextBaseline; textAlign?: CanvasTextAlign }
  export interface ImgStyle extends CursorStyle, BoxStyle, LayoutItemStyle { src: string; x?: Exclude<Length, 'auto'>; y?: Exclude<Length, 'auto'>; stroke?: StrokeStyle; strokeWidth?: number; opacity?: number; alignToPixelGrid?: PixelGridAlignment }
  export interface GStyle extends BoxStyle, LayoutItemStyle { scaleX?: number; scaleY?: number; rotationDegrees?: number; rotationOrigin?: [number, number]; translateX?: number; translateY?: number; skewX?: number; skewY?: number }
  export interface ClipStyle extends BoxStyle, LayoutItemStyle { d: Path2D; fillRule?: CanvasFillRule }
  export interface LayerStyle extends BoxStyle, LayoutItemStyle {}

  export interface FlexProps { children: CRCNode[] }
  export type GridTrack = number | `${number}%` | `${number}fr`;
