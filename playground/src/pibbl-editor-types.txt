/** Options for capturing committed renders of one live Pibbl root.
 * @see {@link PibblFrameRecordingSession}
 */
export interface PibblFrameRecordingOptions {
  /** Recording length in milliseconds, measured after initialization. */
  duration: number;
  /** Maximum capture frequency; defaults to 60 frames per second. */
  frameRate?: number;
  /** Initial estimated memory multiplier; defaults to 1.15. Memory may grow. */
  memoryHeadroom?: number;
  /** Aborting discards the recording, including during initialization. */
  signal?: AbortSignal;
}

/** An active capture; stop keeps frames and cancel discards them.
 * @see {@link PibblFrameRecording}
 */
export interface PibblFrameRecordingSession {
  /** Resolves after frame zero has been captured and capture has started. */
  readonly ready: Promise<void>;
  /** Resolves when retained frames are ready, or rejects on cancellation/failure. */
  readonly finished: Promise<PibblFrameRecording>;
  /** Finish at the current elapsed time, retaining captured frames. Idempotent. */
  stop(): void;
  /** Discard an active recording and release its resources. Idempotent. */
  cancel(): void;
}

/** Metadata for one immutable RGBA bitmap in physical backing pixels.
 * @see {@link PibblRecordedFrame}
 */
export interface PibblRecordedFrameInfo {
  /** Milliseconds since capture began; the initial frame is timestamp zero. */
  readonly timestamp: number;
  /** Bitmap width in physical backing pixels. */
  readonly width: number;
  /** Bitmap height in physical backing pixels. */
  readonly height: number;
}

/** A caller-owned copy of one frame's RGBA8 bytes.
 * @see {@link PibblRecordedFrameInfo}
 */
export interface PibblRecordedFrame extends PibblRecordedFrameInfo {
  /** Independent RGBA8 bytes; mutations never affect retained pixels. */
  readonly data: Uint8ClampedArray<ArrayBuffer>;
}

/** Retained worker-owned frames; explicitly dispose after use.
 * @see {@link PibblFrameRecordingSession}
 */
export interface PibblFrameRecording {
  /** Actual recording length in milliseconds, including the final still hold. */
  readonly duration: number;
  /** Immutable metadata ordered by ascending timestamp. */
  readonly frames: readonly PibblRecordedFrameInfo[];
  /** Copy one frame without transferring ownership of stored pixels.
   * @param index Zero-based recorded frame index.
   * @returns A caller-owned pixel copy and its metadata.
   */
  readFrame(index: number): Promise<PibblRecordedFrame>;
  /** Release the worker, stored pixels, and pending reads/exports. Idempotent. */
  dispose(): void;
}

/** GIF encoding options; encoding never changes the retained recording.
 * @see {@link PibblFrameRecording}
 */
export interface PibblGifExportOptions {
  /** Output width in pixels; omitted preserves source width or the specified height's aspect ratio. */
  width?: number;
  /** Output height in pixels; omitted preserves source height or the specified width's aspect ratio. */
  height?: number;
  /** Maximum palette entries, including transparency when no background is supplied; integer 2–256, default 256. */
  maxColors?: number;
  /** Exact opaque CSS palette colors. Replaces automatic palette selection; must fit maxColors. */
  palette?: readonly string[];
  /** Apply Floyd–Steinberg error diffusion; defaults to false. */
  dithering?: boolean;
  /** NeuQuant sampling factor, integer 1–30; 1 spends most effort, default 10. Ignored with a supplied palette. */
  quantizationSampleFactor?: number;
  /** Maximum pixels sampled for automatic palette selection across up to 32 frames; default 262144. Ignored with a supplied palette. */
  paletteSampleCount?: number;
  /** Output sampling frequency; defaults to 25. Must be at most 100. */
  frameRate?: number;
  /** Zero loops forever; omitted plays once; positive values repeat that many times. */
  repeatCount?: number;
  /** Optional opaque CSS color used to flatten partial alpha before quantization. */
  background?: string;
  /** Abort encoding while preserving the recording for another export. */
  signal?: AbortSignal;
  /** Progress from zero to one, delivered between bounded worker batches.
   * @param progress Fraction of output frames encoded.
   */
  onProgress?: (progress: number) => void;
}

  import type { Signal } from '@pibbl/core';

  /** An event accepted by a machine. */
  export interface MachineEvent { readonly type: string; }
  /** Routes each event discriminator to its narrowed transition. */
  export type MachineEventRoutes<Context, Event extends MachineEvent, Input, State extends string> = {
    readonly [Type in Event['type']]?: MachineTransition<Context, Extract<Event, { readonly type: Type }>, Input, State>;
  };
  /** Lifecycle state of a machine actor. */
  export type MachineStatus = 'not-started' | 'active' | 'stopped' | 'error';
  /** Atomically published state value, immutable context, and actor status. */
  export interface MachineSnapshot<Context, State extends string = string> {
    readonly value: State;
    readonly context: Readonly<Context>;
    readonly status: MachineStatus;
    readonly error: unknown | undefined;
    matches(state: State): boolean;
  }
  /** Cleanup for a callback task when its state exits. */
  export type MachineTaskCleanup = () => void;
  /** Values given to entry, exit, and transition actions. */
  export interface MachineActionContext<Context, Event extends MachineEvent | undefined, Input, SentEvent extends MachineEvent = MachineEvent> {
    readonly context: Readonly<Context>;
    readonly event: Event;
    readonly input: Input;
    send(event: SentEvent): void;
  }
  /** Synchronous effect run by a transition or state lifecycle. */
  export type MachineAction<Context, Event extends MachineEvent | undefined, Input, SentEvent extends MachineEvent = MachineEvent> = (context: MachineActionContext<Context, Event, Input, SentEvent>) => void;
  /** One action or an ordered action list. */
  export type MachineActions<Context, Event extends MachineEvent | undefined, Input, SentEvent extends MachineEvent = MachineEvent> = MachineAction<Context, Event, Input, SentEvent> | readonly MachineAction<Context, Event, Input, SentEvent>[];
  /** Values supplied to one state-owned task. */
  export interface MachineTaskContext<Context, Event extends MachineEvent, Input> {
    readonly context: Readonly<Context>;
    readonly event: Event | undefined;
    readonly input: Input;
    readonly signal: AbortSignal;
    send(event: Event): void;
  }
  /** Cancelable state work. A resolved promise selects its invocation's onDone route. */
  export type MachineTask<Context, Event extends MachineEvent, Input, Output = unknown> = (context: MachineTaskContext<Context, Event, Input>) => void | MachineTaskCleanup | Promise<Output>;
  /** Event sent internally when an invoked promise resolves. */
  export interface MachineTaskDoneEvent<Output = unknown> extends MachineEvent { readonly type: '@pibbl/machine.done'; readonly output: Output; }
  /** Event sent internally when an invoked task fails. */
  export interface MachineTaskErrorEvent extends MachineEvent { readonly type: '@pibbl/machine.error'; readonly error: unknown; }
  /** State-owned task and its optional completion routes. */
  export interface MachineInvocation<Context, Event extends MachineEvent, Input, State extends string> {
    readonly id?: string;
    readonly task: MachineTask<Context, Event, Input>;
    readonly onDone?: MachineTransition<Context, MachineTaskDoneEvent, Input, State, Event>;
    readonly onError?: MachineTransition<Context, MachineTaskErrorEvent, Input, State, Event>;
  }
  /** One guarded transition route. */
  export interface MachineTransitionConfig<Context, Event extends MachineEvent, Input, State extends string, SentEvent extends MachineEvent = Event> {
    readonly target?: State;
    readonly guard?: (context: MachineActionContext<Context, Event, Input, SentEvent>) => boolean;
    readonly actions?: MachineActions<Context, Event, Input, SentEvent>;
    readonly reenter?: boolean;
  }
  /** Target shorthand, one route, or first-match guarded routes. */
  export type MachineTransition<Context, Event extends MachineEvent, Input, State extends string, SentEvent extends MachineEvent = Event> = State | MachineTransitionConfig<Context, Event, Input, State, SentEvent> | readonly MachineTransitionConfig<Context, Event, Input, State, SentEvent>[];
  /** One flat state with lifecycle actions, invoked work, and local event routes. */
  export interface MachineState<Context, Event extends MachineEvent, Input, State extends string> {
    readonly entry?: MachineActions<Context, Event | undefined, Input, Event>;
    readonly exit?: MachineActions<Context, Event, Input, Event>;
    readonly invoke?: MachineInvocation<Context, Event, Input, State> | readonly MachineInvocation<Context, Event, Input, State>[];
    readonly on?: MachineEventRoutes<Context, Event, Input, State>;
  }
  /** Reusable flat-state behavior. Definitions create no actor or scheduled work. */
  export interface MachineDefinition<Context, Event extends MachineEvent, Input = undefined, State extends string = string> {
    readonly initial: State;
    readonly context: Context | ((context: Readonly<{ input: Input }>) => Context);
    readonly states: Readonly<Record<State, MachineState<Context, Event, Input, State>>>;
    readonly on?: MachineEventRoutes<Context, Event, Input, State>;
  }
  /** Input supplied when creating an explicitly owned actor. */
  export interface CreateMachineOptions<Input> { readonly input: Input; }
  /** Explicit machine execution with a readonly signal snapshot. */
  export interface MachineActor<Context, Event extends MachineEvent, _Input, State extends string = string> {
    readonly snapshot: Signal<MachineSnapshot<Context, State>>;
    start(): void;
    send(event: Event): void;
    stop(): void;
  }
  /** Creates an action that shallowly merges a context patch. */
  export function assign<Context extends object, Event extends MachineEvent, Input>(update: (context: MachineActionContext<Context, Event, Input>) => Partial<Context>): MachineAction<Context, Event, Input>;
  /** Marks callback-style cancelable work as a state-owned task. */
  export function fromCallback<Context, Event extends MachineEvent, Input>(callback: MachineTask<Context, Event, Input>): MachineTask<Context, Event, Input>;
  /** Validates and returns a reusable machine definition without creating an actor. */
  export function defineMachine<Context, Event extends MachineEvent, Input = undefined, State extends string = string>(definition: MachineDefinition<Context, Event, Input, State>): MachineDefinition<Context, Event, Input, State>;
  /** Creates an idle explicitly owned actor; call start and stop to control its lifetime. */
  export function createMachine<Context, Event extends MachineEvent, Input, State extends string = string>(definition: MachineDefinition<Context, Event, Input, State>, options: CreateMachineOptions<Input>): MachineActor<Context, Event, Input, State>;
  /** Creates a component-owned actor that starts after mount and stops on unmount. */
  export function useMachine<Context, Event extends MachineEvent, Input, State extends string>(definition: MachineDefinition<Context, Event, Input, State>, options: { readonly input: Input }): MachineActor<Context, Event, Input, State>;
